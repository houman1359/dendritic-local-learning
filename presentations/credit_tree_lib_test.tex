% 13 — heroeq: the factorization (starred = \Large), ADJACENT terms
\begin{frame}{heroeq: factorization with adjacent term annotations}
  \heroeq*[theorem]{%
    \frac{\partial\mathcal L}{\partial g_i}
    \;=\;
    \term[Shunt]{e_i\vphantom{\delta_n}}{local eligibility}\,
    \term[Add]{\delta_n}{transported error}
    \qquad
    \delta_n=\delta_{0,u}\,\widetilde{\alpha}_n}
  \vfill
  {\footnotesize\color{Mute}Adjacent-term collision test: the two braces and
   their labels must stay clear of each other, baselines aligned.\par}
\end{frame}

% 14 — heroeq: the descent bound U(M) with three semantic terms
\begin{frame}{heroeq: descent bound, three-term anatomy}
  \heroeq[bound]{%
    \E\,\mathcal L(\bm w^{+})
    \;\le\;
    \mathcal L(\bm w)
    \;-\;\eta\,\term[Shunt]{\bm g^{\mathsf T}\!M\bm g}{retained signal}
    \;+\;\frac{L\eta^{2}}{2}
      \Bigl(\term[Amber]{\lVert M\bm g\rVert_2^{2}}{curvature bias}
      \;+\;\term[BP]{\tr(M\Sigma M^{\mathsf T})}{admitted noise}\Bigr)}
  \vfill
  {\footnotesize\color{Mute}Eyebrow tag, Shunt left rule only (no box),
   colors per semantics: green retained, amber bias, BP-red noise.\par}
\end{frame}
